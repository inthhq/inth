import { getConsentClient, openDialog } from '@inth/astro/client';
import { InthProvider, useConsent } from '@inth/react';
import { useEffect, useState } from 'react';

const MeasurementStatus = () => {
	const allowed = useConsent('measurement');
	return (
		<p>
			<span data-testid="measurement-status">
				Measurement is {allowed ? 'allowed' : 'not allowed'}.
			</span>{' '}
			<button
				onClick={() => {
					void openDialog();
				}}
				type="button"
			>
				Change cookie settings
			</button>
		</p>
	);
};

const ConsentStatus = () => {
	// The page's consent runtime, which the Inth integration started. For a
	// visitor the banner asks to choose, it loads after the page.
	const [runtime, setRuntime] = useState(() => getConsentClient()?.runtime);
	useEffect(() => {
		const client = getConsentClient();
		if (runtime || !client) {
			return;
		}
		let active = true;
		void (async () => {
			await client.ready;
			if (active) {
				setRuntime(client.runtime);
			}
		})();
		return () => {
			active = false;
		};
	}, [runtime]);
	if (!runtime) {
		return null;
	}
	return (
		// `colorScheme: null` leaves dark mode to the integration.
		<InthProvider options={{ colorScheme: null }} runtime={runtime}>
			<MeasurementStatus />
		</InthProvider>
	);
};

export default ConsentStatus;
