import {
	ChangeDetectionStrategy,
	Component,
	computed,
	inject,
} from '@angular/core';
import { ConsentManager, Portal } from '@inth/angular';
import {
	DialogBackdrop,
	DialogClose,
	DialogContent,
	DialogDescription,
	Dialog,
	DialogPositioner,
	DialogTitle,
	SwitchControl,
	Switch,
	SwitchThumb,
} from '@inth/angular/primitives';

@Component({
	changeDetection: ChangeDetectionStrategy.OnPush,
	imports: [
		DialogBackdrop,
		DialogClose,
		DialogContent,
		DialogDescription,
		Dialog,
		DialogPositioner,
		DialogTitle,
		Portal,
		SwitchControl,
		Switch,
		SwitchThumb,
	],
	selector: 'app-custom-dialog',
	template: `
		<ng-container
			inthDialog
			#dialog="inthDialog"
			[open]="open()"
			(openChange)="setOpen($event)"
			[preventScroll]="true"
			[unmountOnExit]="true"
		>
			@if (dialog.shouldRender()) {
				<div inthPortal>
					<div inthDialogBackdrop class="backdrop"></div>
					<div inthDialogPositioner class="positioner">
						<div inthDialogContent class="card">
							<h2 inthDialogTitle>Cookie settings</h2>
							<p inthDialogDescription>Choose what this site may use.</p>
							<label>
								<button
									inthSwitch
									[checked]="consent.draft.values().measurement ?? false"
									(checkedChange)="consent.setConsent('measurement', $event)"
								>
									<span inthSwitchControl><span inthSwitchThumb></span></span>
								</button>
								Analytics
							</label>
							<button type="button" (click)="consent.saveConsents('custom')">
								Save
							</button>
							<button inthDialogClose>Close</button>
						</div>
					</div>
				</div>
			}
		</ng-container>
	`,
})
export class CustomDialog {
	protected readonly consent = inject(ConsentManager);

	// Open while the consent manager's active surface is the dialog.
	protected readonly open = computed(
		() => this.consent.activeUI() === 'dialog'
	);

	protected setOpen(open: boolean): void {
		this.consent.setActiveUI(open ? 'dialog' : 'none');
	}
}
