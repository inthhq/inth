import {
	ChangeDetectionStrategy,
	Component,
	computed,
	effect,
	inject,
} from '@angular/core';
import { ConsentManager } from '@inth/angular';

@Component({
	changeDetection: ChangeDetectionStrategy.OnPush,
	selector: 'app-consent-status',
	template: `
		@if (consent.hasPolicy()) {
			<p>Analytics: {{ measurement() ? 'on' : 'off' }}</p>
			@if (!consent.explicitChoice()) {
				<p>You have not chosen yet.</p>
			}
			<button type="button" (click)="consent.setActiveUI('dialog')">
				Change cookie settings
			</button>
		}
	`,
})
export class ConsentStatus {
	protected readonly consent = inject(ConsentManager);

	// `has()` reads a signal, so this recomputes when consent changes.
	protected readonly measurement = computed(() =>
		this.consent.has('measurement')
	);

	constructor() {
		effect(() => {
			console.info('Permissions', this.consent.effectivePermissions());
		});
	}
}
