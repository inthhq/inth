import {
	ChangeDetectionStrategy,
	Component,
	computed,
	inject,
} from '@angular/core';
import {
	ConsentManager,
	HeadlessConsent,
	resolveActionLabels,
	resolvePromptCopy,
} from '@c15t/angular';

@Component({
	changeDetection: ChangeDetectionStrategy.OnPush,
	selector: 'app-custom-consent-banner',
	styles: `
		.custom-consent-banner {
			position: fixed;
			inset: auto 1rem 1rem;
			max-width: 32rem;
			padding: 1rem;
			border: 1px solid currentColor;
			border-radius: 0.5rem;
			background: white;
		}
		.custom-consent-banner div {
			display: flex;
			gap: 0.5rem;
			margin-top: 0.5rem;
		}
	`,
	template: `
		@if (headless.banner().isVisible) {
			<section class="custom-consent-banner" [attr.aria-label]="copy().title">
				<h2>{{ copy().title }}</h2>
				<p>{{ copy().description }}</p>
				<!-- The policy decides which actions this visitor gets, and in what order. -->
				@for (group of headless.banner().actionGroups; track $index) {
					<div>
						@for (action of group; track action) {
							<button
								type="button"
								[attr.data-primary]="
									headless.banner().primaryActions.includes(action) || null
								"
								(click)="headless.performAction(action)"
							>
								{{ labels()[action] }}
							</button>
						}
					</div>
				}
			</section>
		}
	`,
})
export class CustomConsentBanner {
	protected readonly headless = inject(HeadlessConsent);
	readonly #consent = inject(ConsentManager);

	// The active language, with English for any missing key.
	readonly #bundle = computed(() => this.#consent.translations()?.translations);
	protected readonly copy = computed(() =>
		resolvePromptCopy(this.#bundle(), this.#consent.policyRule().prompt)
	);
	protected readonly labels = computed(() =>
		resolveActionLabels(this.#bundle())
	);
}
